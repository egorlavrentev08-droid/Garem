-- Создание enum-типов для жёсткой типизации
CREATE TYPE trigger_type AS ENUM (
    '1_DAY_INACTIVE',
    '2_PLUS_DAYS_INACTIVE',
    'ACTIVITY_RESUMED_STREAK',
    'ACTIVITY_RESUMED_AFTER_BREAK'
);

CREATE TYPE intensity_level AS ENUM ('LOW', 'MEDIUM', 'HIGH');

CREATE TYPE mood_type AS ENUM (
    'ANXIOUS',
    'ANGRY',
    'DARK_HUMOR',
    'POSITIVE',
    'MOTIVATIONAL',
    'SARCASTIC',
    'MEME_THREAT',
    'ABSURD',
    'SUPPORTIVE',
    'BLACK_HUMOR',
    'MESSAGING',
    'BIG_STREAK',
    'STRONG',
    'FRIENDLY'
);

-- Основная таблица фраз
CREATE TABLE phrases (
    id SERIAL PRIMARY KEY,
    trigger_type trigger_type NOT NULL,
    intensity intensity_level NOT NULL DEFAULT 'MEDIUM',
    mood mood_type NOT NULL,
    phrase_text TEXT NOT NULL,
    emoji TEXT,
    priority INTEGER DEFAULT 0,
    is_active BOOLEAN DEFAULT true,
    usage_count INTEGER DEFAULT 0,
    last_used TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Индексы для быстрого поиска
CREATE INDEX idx_phrases_trigger ON phrases(trigger_type, intensity, mood);
CREATE INDEX idx_phrases_active ON phrases(is_active);
CREATE INDEX idx_phrases_priority ON phrases(priority DESC);
